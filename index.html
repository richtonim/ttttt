Developer Hub API Reference
quickstart.py 
from noderefine import Client

client = Client(api_key="nr_live_...")

# Connect and analyze a workflow
scan = client.workflows.analyze(
    workflow_id="wf_prod_agent_v2",
    depth="full"
)

for issue in scan.issues:
    print(f"[{issue.severity}] {issue.node}: {issue.message}")

print(f"Health score: {scan.health_score}%")

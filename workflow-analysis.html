workflow-scan.js 
import { NodeRefine } from '@noderefine/sdk';

const client = new NodeRefine({ apiKey: process.env.NR_API_KEY });

const report = await client.workflows.scan({
  workflowId: 'agent-pipeline-v3',
  options: { depth: 'full', includeMetrics: true }
});

console.log(`Health: ${report.healthScore}%`);
console.log(`Issues found: ${report.issues.length}`);
report.issues.forEach(i => console.log(`  [${i.severity}] ${i.nodeId}: ${i.message}`));

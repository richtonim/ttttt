API Overview

import { NextLayer } from "@nextlayer/sdk";

const client = new NextLayer({
  apiKey: process.env.NEXTLAYER_API_KEY
});

const response = await client.agents.run({
  agent: "support-agent",
  input: "Summarize this customer request",
  user: "customer_1024"
});

console.log(response.output);
